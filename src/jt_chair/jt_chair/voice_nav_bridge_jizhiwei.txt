#!/usr/bin/env python3
import rclpy
from rclpy.node import Node
from rclpy.action import ActionClient
import serial
import math
import threading

from nav2_msgs.action import NavigateToPose
from geometry_msgs.msg import Twist
from tf2_ros import Buffer, TransformListener

class VoiceNavBridge(Node):
    def __init__(self):
        super().__init__('voice_nav_bridge')

        # ==========================================
        # 1. 串口硬件配置
        # ==========================================
        self.serial_port = '/dev/ttyS4'
        self.baud_rate = 115200
        try:
            self.ser = serial.Serial(self.serial_port, self.baud_rate, timeout=0.5)
            self.get_logger().info(f"[Hardware] 串口通信链路已建立: {self.serial_port} (Baud: {self.baud_rate})")
        except Exception as e:
            self.get_logger().error(f"[Hardware] 串口初始化失败，请验证硬件连线与系统权限: {e}")
            return

        # ==========================================
        # 2. 目标坐标系与指令映射矩阵
        # ==========================================
        # 依据点 B (马桶位) 向前推算 0.8m 作为点 A (准备位)
        self.location_map = {
            0x32: {"name": "客厅", "pose": [-1.687, 4.518, -0.013]},
            0x33: {"name": "厨房", "pose": [4.910, 2.084, 0.031]},
            0x34: {"name": "卧室", "pose": [0.000, 0.000, 0.000]},
            # 0x31: 仅执行全局导航至准备点 A，无后续控制交接
            # 原始厕所坐标 0x31: {"name": "厕所(仅到达)", "pose": [3.206, -0.296, 1.599]}, 
            0x31: {"name": "厕所(仅到达)", "pose": [-0.198, 0.014, 0.024]}, 
            # 0x35: 全局导航至点 A 完毕后，触发状态机切换，接管控制权倒退至点 B
            # 原始高级厕所坐标 0x35: {"name": "厕所倒车模式", "pose": [3.206, -0.296, 1.599]} 
            0x35: {"name": "厕所倒车模式", "pose": [-0.198, 0.014, 0.024]} 
        }

        # ==========================================
        # 3. 闭环 PID 泊车控制参数矩阵 (点 B：马桶基准坐标)
        # ==========================================
        #原始马桶坐标
        #self.toilet_dock_target = [3.206, -0.796, 1.599] 
        self.toilet_dock_target =  [-1.878, -0.296, 0.024] #新马桶测试

        # 到达准备点后，向语音终端反馈的十六进制同步信号
        self.feedback_hex = 0x01 
        
        self.Kp_yaw = 0.6  
        self.Kd_yaw = 0.3  
        self.prev_yaw_error = 0.0

        # ROS 2 接口定义
        self.cmd_pub = self.create_publisher(Twist, '/cmd_vel', 10)
        self.nav_client = ActionClient(self, NavigateToPose, 'navigate_to_pose')
        
        # 引入 TF2 空间变换监听器，实现基于全局坐标系的绝对定位
        self.tf_buffer = Buffer()
        self.tf_listener = TransformListener(self.tf_buffer, self)
        
        # 创建 20Hz 控制循环，专门用于泊车时的高频状态监控与指令下发
        #self.docking_timer = self.create_timer(0.05, self.docking_control_loop)
        
        # 系统状态机与任务锁
        self.state = 'IDLE'           
        self.current_target_name = "" 
        self.is_waiting = False       

        # 挂载异步监听线程
        self.read_thread = threading.Thread(target=self.serial_loop)
        self.read_thread.daemon = True 
        self.read_thread.start()
        self.get_logger().info("[System] 异步语音监听守护线程已启动。")

    # ==========================================================
    # 异步通信解析与状态机流转逻辑
    # ==========================================================
    def serial_loop(self):
        while rclpy.ok():
            try:
                if self.ser.in_waiting > 0:
                    incoming_byte = self.ser.read(1)
                    if not incoming_byte:
                        continue
                    
                    cmd_hex = ord(incoming_byte)
                    
                    # 硬件级底噪过滤机制
                    if cmd_hex in [0xff, 0x00]:
                        continue

                    self.get_logger().info(f"[Serial] 接收到有效载荷 (Hex): {hex(cmd_hex)}")

                    if cmd_hex in self.location_map:
                        self.state= 'IDLE'
                        # 状态机互斥锁校验 1：执行态排他
                        if self.state != 'IDLE':
                            self.get_logger().warn("[State] 状态机处于非空闲态 (Active)，当前指令已被系统丢弃。")
                            self.ser.reset_input_buffer()
                            continue
                            
                        # 状态机互斥锁校验 2：防抖延时排他
                        if self.is_waiting:
                            self.get_logger().warn("[State] 系统处于任务延时触发周期，防抖机制已拦截重复指令。")
                            self.ser.reset_input_buffer()
                            continue

                        target = self.location_map[cmd_hex]
                        self.current_target_name = target['name']
                        self.state = 'NAVIGATING'
                        self.is_waiting = True
                        
                        self.get_logger().info(f"----------------------------------------")
                        self.get_logger().info(f"[Trigger] 指令解析成功。目标分配: {self.current_target_name}")
                        self.get_logger().info(f"[System] 挂起导航任务，进入启动前保护期 (3.0s)...")
                        self.get_logger().info(f"----------------------------------------")
                        
                        def execute_delayed_nav():
                            self.get_logger().info("[System] 保护期结束，向 Nav2 服务器发起目标规划请求。")
                            self.send_goal(target['pose'])
                            self.is_waiting = False
                            self.ser.reset_input_buffer() 

                        timer = threading.Timer(3.0, execute_delayed_nav)
                        timer.start()

                    else:
                        self.get_logger().warn(f"[Parser] 未定义指令序列: {hex(cmd_hex)}，解析器拒绝响应。")
                        
            except Exception as e:
                self.get_logger().error(f"[Serial] 串行数据流读取引发异常: {e}")
                import time
                time.sleep(1.0)

    # ==========================================================
    # Nav2 全局规划控制层
    # ==========================================================
    def send_goal(self, coords):
        if not self.nav_client.wait_for_server(timeout_sec=10.0):
            self.get_logger().error("[Action] Nav2 服务器连接超时，请验证导航栈健康状态。")
            self.state = 'IDLE'
            return

        goal_msg = NavigateToPose.Goal()
        goal_msg.pose.header.frame_id = 'map'
        goal_msg.pose.header.stamp = self.get_clock().now().to_msg()
        goal_msg.pose.pose.position.x = coords[0]
        goal_msg.pose.pose.position.y = coords[1]
        
        yaw = coords[2]
        goal_msg.pose.pose.orientation.z = math.sin(yaw / 2.0)
        goal_msg.pose.pose.orientation.w = math.cos(yaw / 2.0)

        self.send_goal_future = self.nav_client.send_goal_async(goal_msg, feedback_callback=self.feedback_callback)
        self.send_goal_future.add_done_callback(self.goal_response_callback)

    def goal_response_callback(self, future):
        goal_handle = future.result()
        if not goal_handle.accepted:
            self.get_logger().warn('[Action] 目标点被规划器拒绝 (异常原因: 代价地图重叠或拓扑不可达)。')
            self.state = 'IDLE'
            return
        self.get_logger().info('[Action] 目标位姿已由 Nav2 接受，路径追踪器介入。')
        self.goal_result_future = goal_handle.get_result_async()
        self.goal_result_future.add_done_callback(self.goal_result_callback)

    def feedback_callback(self, feedback_msg):
        pass # 静默高频的位姿反馈数据

    
    def goal_result_callback(self, future):
        status = future.result().status
        
        # ROS 2 Action Status 4 标识 SUCCEEDED
        if status == 4: 
            self.get_logger().info('[Action] 全局导航阶段成功收敛 (Status: SUCCEEDED)。')
            
            if self.current_target_name == "厕所倒车模式":
                # 阶段同步逻辑：向语音终端下发触达反馈
                self.get_logger().info(f"[Sync] 正在向语音终端下发同步载荷: {hex(self.feedback_hex)}")
                try:
                    self.ser.write(bytes([self.feedback_hex]))
                except Exception as e:
                    self.get_logger().error(f"[Sync] 串口下行链路发送失败: {e}")

                self.get_logger().info("[System] 开启泊车前时滞挂起周期 (3.0s)...")
                
                # 定义 3 秒后的控制权交接回调
                def start_docking_sequence():
                    self.get_logger().info("[Control] 时滞结束。接管底盘运动学控制权，激活闭环 PID 泊车序列。")
                    self.state = 'DOCKING'
                    self.prev_yaw_error = 0.0

                threading.Timer(3.0, start_docking_sequence).start()
            else:
                self.get_logger().info("[System] 导航序列完成。系统状态机复位至 IDLE 态。")
                self.state = 'IDLE'
        elif status == 6:
            self.get_logger().warn('[Action] 任务被规划器中止 (Status: ABORTED)，可能路径被动态障碍物截断。')
            self.state = 'IDLE'
        elif status == 5:
            self.get_logger().warn('[Action] 任务已被外部控制器取消 (Status: CANCELED)。')
            self.state = 'IDLE'
        else:
            self.get_logger().warn(f'[Action] 任务异常终止，未记录的状态码: {status}。')
            self.state = 'IDLE'

    # 基于 TF2 全局绝对坐标的闭环泊车控制层 (20Hz)
# ==========================================================
def docking_control_loop(self):
    # 状态屏蔽，非泊车阶段直接挂起
    if self.state != 'DOCKING':
        return
        
    try:
        trans = self.tf_buffer.lookup_transform('map', 'base_link', rclpy.time.Time())
    except Exception as e:
        return
        
    current_x = trans.transform.translation.x
    current_y = trans.transform.translation.y
    
    q = trans.transform.rotation
    siny_cosp = 2 * (q.w * q.z + q.x * q.y)
    cosy_cosp = 1 - 2 * (q.y * q.y + q.z * q.z)
    current_yaw = math.atan2(siny_cosp, cosy_cosp)

    distance = math.hypot(self.toilet_dock_target[0] - current_x, self.toilet_dock_target[1] - current_y)
    
    self.get_logger().info(
        f"距离目标点: {distance:.2f} 米 | "
        f"当前坐标: [{current_x:.2f}, {current_y:.2f}] | "
        f"目标坐标: [{self.toilet_dock_target[0]:.2f}, {self.toilet_dock_target[1]:.2f}]"
    )

    if distance < 0.15:
        self.stop_robot()
        self.state = 'IDLE' 
        self.get_logger().info("[Control] 泊车序列执行完毕")
        return

    # ========== 新增：Y轴横向位置闭环，让Y往-0.3靠 ==========
    y_pos_error = self.toilet_dock_target[1] - current_y
    Kp_y_pos = 0.3  # 横向纠偏强度，太大容易抖，太小靠不过去
    yaw_correct_from_y = Kp_y_pos * y_pos_error

    target_yaw = self.toilet_dock_target[2] + yaw_correct_from_y  # 航向叠加横向纠偏
    yaw_error = math.atan2(math.sin(target_yaw - current_yaw), math.cos(target_yaw - current_yaw))
    
    if abs(yaw_error) < 0.03:
        yaw_error = 0.0

    d_error = yaw_error - self.prev_yaw_error
    raw_angular_z = (self.Kp_yaw * yaw_error) + (self.Kd_yaw * d_error)
    self.prev_yaw_error = yaw_error

    angular_z = raw_angular_z
    linear_x = -0.15 
    
    if distance < 0.40:
        linear_x = -0.04 - (distance / 0.40) * 0.11
        angular_z = raw_angular_z * (distance / 0.40)
        
    angular_z = max(-0.4, min(0.4, angular_z))

    cmd = Twist()
    cmd.linear.x = linear_x
    cmd.angular.z = angular_z 
    
    self.cmd_pub.publish(cmd)


def main(args=None):
    rclpy.init(args=args)
    node = VoiceNavBridge()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.stop_robot()
        node.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()
